pragma solidity ^0.8.34;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/token/ERC721/IERC721.sol";

contract FarmCandyCane is ERC721 {
    IERC721 public immutable farmNFT;
    uint256 public nextTokenId = 1;

    // Дедлайн: 30 января 2027, 23:59:59 UTC
    // 2027-01-30 23:59:59 UTC = 1830470399 (unix timestamp)
    uint256 public immutable claimDeadline;

    string private constant METADATA_CID = "bafkreihyxjgb4bepkd3leb3546thet6ofn55dywc7juudjn7yi3pbmgpve";

    mapping(address => bool) public hasMinted;

    constructor(address _farmNFT, uint256 _claimDeadline) ERC721("Candy Cane 2027", "CCN27") {
        farmNFT = IERC721(_farmNFT);
        claimDeadline = _claimDeadline;
    }

    function mint() external {
        require(block.timestamp <= claimDeadline, "Claim period ended");
        require(farmNFT.balanceOf(msg.sender) >= 1, "Not a Farm holder");
        require(!hasMinted[msg.sender], "Already minted");

        hasMinted[msg.sender] = true;
        uint256 tokenId = nextTokenId++;
        _safeMint(msg.sender, tokenId);
    }

    function tokenURI(uint256 tokenId) public view override returns (string memory) {
        require(tokenId >= 1 && tokenId < nextTokenId, "Nonexistent token");
        return string.concat("ipfs://", METADATA_CID);
    }
}
